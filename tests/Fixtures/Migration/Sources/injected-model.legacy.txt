<?php
use Pixie\QueryBuilder\QueryBuilder;
use Pixie\QueryBuilder\QueryBuilderHandler;
final class SyntheticStaffModel
{
    public function __construct(private QueryBuilderHandler $database) {}
    public function active(): array
    {
        return $this->database->table('staff', 's')
            ->select(['s.id', $this->database->raw('UPPER(s.name) AS display')])
            ->join('departments', 'd.id', '=', 's.department_id')
            ->where('s.active', true)
            ->get();
    }
    public function pdo(): PDO
    {
        return $this->database->getConnection()->getPdoInstance();
    }
}
