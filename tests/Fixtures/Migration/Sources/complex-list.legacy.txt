<?php
use Pecee\Pixie\Connection;
use Pecee\Pixie\QueryBuilder\QueryBuilderHandler;
final class SyntheticMemberList
{
    public function __construct(private Connection $database) {}
    private function db(): QueryBuilderHandler
    {
        return new QueryBuilderHandler($this->database);
    }
    public function page(array $teams): array
    {
        $query = $this->db()->table('members', 'm')
            ->join('teams', 't.id', '=', 'm.team_id')
            ->leftJoin('roles', 'r.id', '=', 'm.role_id')
            ->where(function (QueryBuilderHandler $group): void {
                $group->where('m.active', true)->orWhere('m.status', 'pending');
            })
            ->whereIn('m.team_id', $teams)
            ->orderBy('m.id')
            ->limit(25);
        return [$query->get(), $query->count()];
    }
    public function create(array $row): string
    {
        return $this->db()->table('members')->insert($row);
    }
}
