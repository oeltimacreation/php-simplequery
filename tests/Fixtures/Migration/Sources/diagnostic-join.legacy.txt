<?php
use Pecee\Pixie\Connection;
use Pecee\Pixie\QueryBuilder\QueryBuilderHandler;
final class SyntheticDiagnosticQuery
{
    public function __construct(private Connection $database) {}
    private function db(): QueryBuilderHandler
    {
        return new QueryBuilderHandler($this->database);
    }
    public function run(string $kind): array
    {
        $query = $this->db()->table('users', 'u')
            ->join('assignments', "a.user_id = u.id AND a.kind = '$kind'")
            ->where('u.active', true);
        $rows = $query->get();
        log_sql($query->getLastQuery()->getRawSql());
        return $rows;
    }
}
