<?php

declare(strict_types=1);

use Oeltima\SimpleQuery\Connection;
use Oeltima\SimpleQuery\Exception\ExternalTransactionException;
use Oeltima\SimpleQuery\Exception\QueryExecutionException;

final class SyntheticProductionTransaction
{
    public function __construct(private Connection $database)
    {
    }

    public function externallyOwned(): void
    {
        $pdo = $this->database->pdo();
        $pdo->beginTransaction();
        $this->database->table('jobs')->where('state', 'open')->first();
        $pdo->commit();
    }

    public function sqliteImmediate(): void
    {
        $pdo = $this->database->pdo();
        $pdo->exec('BEGIN IMMEDIATE');
        $this->database->table('locks')->where('key', 'synthetic')->update(['held' => true]);
        $pdo->rollBack();
    }

    public function nestedManaged(): void
    {
        $this->database->transaction(function (): void {
            $this->database->transaction(function (): void {
                $this->database->table('jobs')->delete();
            });
        });
    }

    public function refusesExternalOwnership(): bool
    {
        try {
            $this->database->transaction(static function (): void {
            });
        } catch (ExternalTransactionException) {
            return true;
        }

        return false;
    }

    public function isLockConflict(QueryExecutionException $exception): bool
    {
        return in_array($exception->driverCode, ['1205', '1213', '5', '6'], true);
    }
}
